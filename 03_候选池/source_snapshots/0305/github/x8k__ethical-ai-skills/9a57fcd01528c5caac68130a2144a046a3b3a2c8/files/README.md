## 🛠️ **Technical Integration for AI Developers**

### Programmatic Access
```python
from pathlib import Path
import json

class EthicalSkillLoader:
    def __init__(self, repo_path="./skills"):
        self.repo_path = Path(repo_path)
        
    def discover_skills(self):
        """Find all available skills in the repository"""
        return [d for d in self.repo_path.iterdir() if d.is_dir()]
    
    def load_skill(self, skill_name):
        """Load a specific skill with all its components"""
        skill_path = self.repo_path / skill_name
        
        # Load metadata
        with open(skill_path / "metadata.json") as f:
            metadata = json.load(f)
        
        # Load knowledge base
        knowledge = self._load_knowledge(skill_path / "knowledge")
        
        # Load prompts
        prompts = self._load_prompts(skill_path / "prompts")
        
        # Load evaluation cases
        eval_cases = self._load_eval_cases(skill_path / "eval")
        
        return {
            "metadata": metadata,
            "knowledge": knowledge,
            "prompts": prompts,
            "eval_cases": eval_cases
        }
    
    def _load_knowledge(self, path):
        """Load all knowledge files from a skill"""
        knowledge = {}
        for f in path.glob("*.md"):
            knowledge[f.stem] = f.read_text()
        return knowledge
```

    def _prepare_dataset(self, skill_data):

                    "input": chunk,
                    "output": self._generate_ideal_response(chunk, skill_data)
                })
